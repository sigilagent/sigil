
# ---- bundle on path: the skill's owned modules (core/, scripts/, …) are copied
# into the run dir; make `from core.x import y` resolve for in-process tools and
# the authored scripts they exec. Runs once at agent import. ----
def _ensure_path():
    import os, sys, shutil
    here = ""
    try:
        here = os.path.dirname(os.path.abspath(__file__))
    except Exception:
        here = ""
    for d in (os.getcwd(), here):
        if d and d not in sys.path:
            sys.path.insert(0, d)
    # Stage the skill's owned bundle resources (templates/, core/, scripts/, *.md,
    # requirements.txt, …) from the agent dir into cwd, so the agent's relative file
    # reads (open("templates/viewer.html")) and module imports (from core import x)
    # resolve when it runs in a fresh bench workdir rather than inside the skill bundle.
    cwd = os.getcwd()
    # NEVER stage from a system temp root: a module placed directly in /tmp
    # would "stage" every /tmp resident (browser locks, other sessions' dirs)
    # into the invoking cwd. Bundles live in their own directories.
    _tmp_roots = {"/tmp", "/private/tmp", "/var/tmp",
                  os.path.realpath(os.environ.get("TMPDIR", "/nonexistent")).rstrip("/")}
    if os.path.realpath(here or "/nonexistent").rstrip("/") in _tmp_roots:
        return
    # A git root is never a skill bundle: an eject living in a real repo (e.g.
    # ~/.sigil/app) would stage the WHOLE repo into the invoking cwd (observed
    # live: a fresh sandbox swallowed the entire sigil checkout). Bundle dirs
    # (skill folders with core/, scripts/, templates/) have no .git of their own.
    if here and os.path.isdir(os.path.join(here, ".git")):
        return
    if here and here != cwd and os.path.isdir(here):
        # Only SKILL RESOURCES cross into cwd. The bundle dir is also where a
        # host wrapper keeps its own things — run.sh, a .env file with live
        # keys, a private HOME/cache for the runtime — and staging those put
        # the keys into a graded task workspace and copytree'd a 828M runtime
        # tree (symlink followed) into it on every boot (caught live:
        # SkillsBench MCP cells). Dot-entries, wrappers, env files, homes and
        # caches never stage; symlinks stage AS symlinks.
        _skip = {"agent.jac", "voodoo-bench.toml", ".jac", "__pycache__",
                 "agent_result.txt", "res", "SKILL.md", "agent.ir", "LICENSE.txt",
                 "run.sh", "home", "cache", "node_modules"}
        for _name in os.listdir(here):
            if _name in _skip or _name.startswith(".") or _name.endswith((".jir", ".pyc", ".bak", ".log")):
                continue
            if _name.startswith(("agent.", "build.", "compile.")) and _name != "agent.jac":
                continue  # compile provenance (agent.agir, build.json, compile.log) is not a resource
            _src = os.path.join(here, _name); _dst = os.path.join(cwd, _name)
            if os.path.exists(_dst) or os.path.islink(_dst):
                continue
            try:
                if os.path.islink(_src):
                    os.symlink(os.readlink(_src), _dst)
                elif os.path.isdir(_src):
                    shutil.copytree(_src, _dst, symlinks=True)
                else:
                    shutil.copy2(_src, _dst)
            except Exception:
                pass

_ensure_path()

# ---- runtime hardening switches: on by default, SIGIL_<NAME>=0 turns one off ----
# Same contract as the compiler's gate flags: 0/false/no/off disable; anything
# else, including unset, leaves the behaviour on. An escape hatch for a task the
# heuristic is wrong about, not a tuning knob.
def _sigil_on(name):
    import os
    return str(os.environ.get("SIGIL_" + str(name).upper(), "")).strip().lower() not in ("0", "false", "no", "off")

# ---- live progress: append a node-entry event to AGIR_PROGRESS (one JSON/line) ----
def _progress(node):
    import os, json, time
    # ---- step budget: a walker cycling a graph (an unanswered HIL loop, a bad
    # route) must abort LOUDLY, not spin to the harness timeout. Budget is
    # env-tunable; the abort names the node and the visit count. ----
    g = globals()
    g["_NVISITS"] = g.get("_NVISITS", 0) + 1
    budget = int(os.environ.get("AGIR_STEP_BUDGET", "256") or 256)
    if g["_NVISITS"] > budget:
        raise RuntimeError(
            "step budget exhausted: %d node visits (budget %d), last node '%s' — "
            "likely a loop with no converging guard; raise AGIR_STEP_BUDGET or fix the route"
            % (g["_NVISITS"], budget, node))
    _tui_node(node)
    p = os.environ.get("AGIR_PROGRESS", "")
    if not p:
        return
    try:
        with open(p, "a") as f:
            f.write(json.dumps({"node": node, "t": time.time()}))
            f.write("\n")
            f.flush()
    except Exception:
        pass

# ---- artifact TUI: a compiled skill run IS a terminal experience. Pure ANSI,
# zero dependencies (the ejected artifact stays self-contained). Auto-on when
# stdout is a TTY; AGIR_TUI=0 forces off, AGIR_TUI=1 forces on (tests). Batch
# and harness runs (captured stdout) are byte-identical to before. ----
_TUI = {"last": None, "t0": None, "start": None}

def _tui_on():
    import os, sys
    v = os.environ.get("AGIR_TUI", "")
    if v == "0":
        return False
    if v == "1":
        return True
    try:
        return sys.stdout.isatty()
    except Exception:
        return False

_V = "\033[38;5;135m"   # violet
_C = "\033[38;5;51m"    # cyan
_D = "\033[2m"          # dim
_R = "\033[0m"

def _tui_banner(name, task):
    import sys, time
    if not _tui_on():
        return
    _TUI["start"] = time.time()
    sys.stdout.write(_V + "\n  \u25c6 " + _R + str(name) + "\n")
    sys.stdout.write(_D + "    " + str(task)[:120] + _R + "\n\n")
    sys.stdout.flush()

def _tui_close(now=None):
    # Finish the in-place node line. A node whose detail already streamed was
    # committed by _tui_sub, so there is nothing left to rewrite.
    import sys, time
    now = now or time.time()
    if _TUI.get("open") and _TUI["last"] is not None:
        dt = now - (_TUI["t0"] or now)
        sys.stdout.write("\r\033[2K  " + _V + "\u25c6 " + _R + str(_TUI["last"])
                         + _D + "  %.1fs" % dt + _R + "\n")
        sys.stdout.flush()
    _TUI["open"] = False

def _fmt_val(v, n=72):
    # One line, collapsed whitespace, elided \u2014 a value preview, never a dump.
    s = " ".join(str(v).split())
    return s[:n] + ("\u2026" if len(s) > n else "")

def _tui_sub(icon, color, text, tail=""):
    # One indented detail line under the current node. The first one commits the
    # node's in-place line (without its duration) so detail can stream beneath.
    import sys
    if not _tui_on():
        return
    if _TUI.get("open"):
        sys.stdout.write("\r\033[2K  " + _V + "\u25c6 " + _R + str(_TUI["last"]) + "\n")
        _TUI["open"] = False
    sys.stdout.write("      " + color + icon + _R + " " + text
                     + ((_D + "  " + tail + _R) if tail else "") + "\n")
    sys.stdout.flush()

def _tui_tool(name, args, fn):
    # Run a tool with the call AND its return visible: what went in, what came
    # back, how long it took. Errors surface as a failed call, then re-raise.
    import time
    if not _tui_on():
        return fn()
    shown = ", ".join("%s=%s" % (k, _fmt_val(v, 24)) for k, v in args if v is not None)
    _tui_sub("\u2699", _C, str(name) + "(" + shown + ")")
    t = time.time()
    try:
        out = fn()
    except Exception as ex:
        _tui_sub("\u2717", "\033[31m", "%s: %s" % (type(ex).__name__, _fmt_val(ex, 56)),
                 "%.1fs" % (time.time() - t))
        raise
    _tui_sub("\u21b3", _D, _fmt_val(out), "%.1fs" % (time.time() - t))
    return out

def _tui_llm_in(slot, reads=None):
    # A `by llm` slot is about to run: which slot, and what it was handed.
    shown = ""
    if reads:
        shown = " " + _D + ", ".join("%s=%s" % (k, _fmt_val(v, 20)) for k, v in reads) + _R
    _tui_sub("\u273b", _V, "llm " + str(slot) + shown)
    _TUI["llm_t"] = __import__("time").time()

def _tui_llm_out(val):
    import time
    dt = time.time() - (_TUI.get("llm_t") or time.time())
    _tui_sub("\u21b3", _D, _fmt_val(val), "%.1fs" % dt)

def _tui_node(node):
    import sys, time
    if not _tui_on():
        return
    now = time.time()
    _tui_close(now)
    sys.stdout.write("  " + _V + "\u25c6 " + _R + str(node) + _D + " \u2026" + _R)
    sys.stdout.flush()
    _TUI["last"] = node
    _TUI["t0"] = now
    _TUI["open"] = True

def _tui_finish(report, artifacts=None):
    import sys, time
    if not _tui_on():
        return
    now = time.time()
    _tui_close(now)
    total = now - (_TUI["start"] or now)
    rep = str(report or "")
    bad = rep.startswith("INCOMPLETE") or not rep
    mark = "\033[31m\u2716" if bad else _C + "\u2714"
    sys.stdout.write("\n  " + mark + _R + _D + "  %.1fs total" % total + _R + "\n")
    sys.stdout.write("  " + ("\u2500" * 56) + "\n")
    for line in (rep or "(no report)").splitlines()[:14]:
        sys.stdout.write("  " + line[:110] + "\n")
    if artifacts:
        sys.stdout.write("  " + ("\u2500" * 56) + "\n  " + _D + "artifacts:" + _R + "\n")
        for a in artifacts[:10]:
            sys.stdout.write("    " + _C + str(a) + _R + "\n")
    sys.stdout.write("\n")
    sys.stdout.flush()

# ---- human-in-the-loop channel: file-based Q&A (sigil chat/approvals attach it
# via HIL_QA_FILE; the superpowers HIL session prototyped this). Batch-safe: with
# no channel attached it returns an HONEST sentinel immediately — never blocks,
# never fabricates a user. ----
def _ask_user(question=None):
    import os, time, json
    qf = os.environ.get("HIL_QA_FILE", "")
    if not qf:
        # in a terminal, the human IS here: ask inline (the compiled skill's
        # clarify loops and review gates become a real conversation)
        if _tui_on():
            import sys
            sys.stdout.write("\r\033[2K\n  " + _C + "? " + _R + str(question) + "\n")
            sys.stdout.flush()
            try:
                return input("  \u203a ")
            except (EOFError, KeyboardInterrupt):
                return "[user closed the prompt]"
        return "[no human channel attached - proceeding without user input]"
    _tui_sub("?", _C, _fmt_val(question, 88))
    af = qf + ".answers"
    n = 0
    try:
        n = sum(1 for _ in open(qf))
    except Exception:
        pass
    try:
        with open(qf, "a") as f:
            f.write(json.dumps({"i": n, "q": str(question)}) + "\n")
            f.flush()
    except Exception:
        return "[human channel write failed]"
    deadline = time.time() + int(os.environ.get("HIL_TIMEOUT", "600") or 600)
    while time.time() < deadline:
        try:
            for line in open(af):
                rec = json.loads(line)
                if rec.get("i") == n:
                    ans = str(rec.get("a", ""))
                    _tui_sub("↳", _D, _fmt_val(ans, 88), "human")
                    return ans
        except Exception:
            pass
        time.sleep(2)
    _tui_sub("✗", "\033[31m", "no answer within HIL_TIMEOUT")
    return "[human did not answer in time]"

# ---- shared OSP runtime helpers (deliver / execute / fence-strip / module read) ----
def _strip_fences(code):
    c = str(code).strip()
    if c.startswith("```"):
        lines = c.splitlines()[1:]
        if lines and lines[-1].strip().startswith("```"):
            lines = lines[:-1]
        c = "\n".join(lines)
    c = _dedent_code(c)
    # A code carry that does not PARSE as-is (glm's planner: "IndentationError:
    # unexpected indent (<string>, line 2)" three times, drone 0.50 vs 0.77) is
    # tried in the shapes a model actually emits: the first fenced block inside
    # prose, the text after a one-line preamble, the whole thing dedented. The
    # first candidate that parses wins; otherwise the text is returned unchanged
    # (a non-Python carry -- a report, JSON -- must never be rewritten here).
    try:
        import ast as _ast, re as _re, textwrap as _tw
        try:
            _mod = _ast.parse(c)
            # a "revise the script" slot answered with the script as a Python string
            # constant (`build_code = ('from openpyxl ...')`): that parses, but running it
            # does nothing. Unwrap when the module is ONE assignment of a string that is
            # itself a Python program (xlsx revise_build_script, both runtimes).
            try:
                if len(_mod.body) == 1 and isinstance(_mod.body[0], _ast.Assign):
                    _v = _mod.body[0].value
                    _inner = None
                    if isinstance(_v, _ast.Constant) and isinstance(_v.value, str):
                        _inner = _v.value
                    elif isinstance(_v, _ast.BinOp):
                        _parts = []
                        def _walk(_n):
                            if isinstance(_n, _ast.BinOp) and isinstance(_n.op, _ast.Add):
                                _walk(_n.left); _walk(_n.right)
                            elif isinstance(_n, _ast.Constant) and isinstance(_n.value, str):
                                _parts.append(_n.value)
                            else:
                                raise ValueError("not a string concat")
                        _walk(_v)
                        _inner = "".join(_parts)
                    if _inner and "\n" in _inner:
                        _ast.parse(_inner)
                        return _inner.strip("\n")
            except Exception:
                pass
            return c
        except SyntaxError:
            pass
        cands = []
        for m in _re.finditer(r"```[a-zA-Z0-9_+-]*[ \t]*\n(.*?)```", str(code), _re.S):
            cands.append(m.group(1))
        lines = c.splitlines()
        if len(lines) > 1:
            cands.append("\n".join(lines[1:]))
        cands.append(_tw.dedent(c))
        for cand in cands:
            cand = _tw.dedent(cand).strip("\n")
            if not cand.strip():
                continue
            try:
                _ast.parse(cand)
                return cand
            except SyntaxError:
                continue
    except Exception:
        pass
    return c

def _dedent_code(src):
    # A GEN slot's code carry can arrive indented as a block ("    import x")
    # — exec() then dies with IndentationError: unexpected indent (caught live:
    # drone-planning-control, position-controller planner_src, both runtimes).
    import textwrap
    t = str(src or "")
    lines = t.splitlines()
    first = next((ln for ln in lines if ln.strip()), "")
    if first[:1] in (" ", "\t"):
        return textwrap.dedent(t)
    return t


def _clean_strlist(v):
    # byLLM's structured output over a small local model leaks JSON structure into
    # list elements: a 5-item answer comes back with a trailing "" or a fragment
    # like '"},{"'. Downstream that is indistinguishable from a real 6th item, so a
    # column-count check fails and the repair slot "fixes" the blank by NAMING it
    # (extra_column, empty_field) instead of dropping it — the gate rejects that too
    # and the route loops until the step budget blows.
    #
    # Drop elements that carry no content. Only whitespace and JSON punctuation
    # qualify, so a legitimately meaningful value is never discarded.
    if not isinstance(v, (list, tuple)):
        return v
    out = []
    for x in v:
        s = str(x if x is not None else "").strip()
        if not s:
            continue
        if not s.strip('{}[](),:;"\'`\\ \t\r\n'):
            continue                       # pure structural punctuation, no content
        out.append(x)
    return out

def _exec_script(script, args=None):
    # The author writes a self-contained script that either writes an output file OR
    # prints the answer to stdout. We run it and return the PATH of the deliverable.
    with open("op.py", "w") as f:
        f.write(_strip_fences(script))
    # use THIS interpreter (sys.executable), not whatever `python3` is on PATH,
    # so the script sees the same installed libs (openpyxl, reportlab, …).
    r = _subprocess.run([sys.executable, "op.py"], capture_output=True, text=True, timeout=180)
    if r.returncode != 0:
        raise RuntimeError(f"script failed: {r.stderr[:800]}")
    # 1) An explicit output file the script was asked to write.
    try:
        p = args.output_spec.path
        if p and os.path.isfile(p):
            return p
    except Exception:
        pass
    # 2) Else the deliverable is whatever the script PRINTED — DON'T discard stdout
    #    (that dropped every print-based answer). Persist it to a real file so the
    #    verifier and the terminal report (_artifact_text) can read the actual answer.
    out = (r.stdout or "").strip()
    if out:
        with open("output.txt", "w") as f:
            f.write(out)
        return "output.txt"
    # 3) Nothing printed and no file — fall back to any spec path, else the script.
    try:
        return args.output_spec.path
    except Exception:
        return "op.py"

# Side-channel for tool-produced data: a tools-ReAct slot returns a final TEXT
# message (often not the script), but the real product was built inside a tool
# call. Tools stash their product here; the author node retrieves it. Per-run
# process, so it starts empty (no cross-run leakage).
def _stash(key, value):
    globals().setdefault("_OSP_PRODUCT", {})[key] = value
    return value

def _unstash(key):
    v = globals().get("_OSP_PRODUCT", {}).get(key)
    return v if v else []

def _exec_var(script, varname):
    # Execute an authored script and return the named product (e.g. `frames`) to
    # thread into downstream OSP nodes. Accept either a bare `<varname>` list, a
    # build_<varname>() factory, OR an object that accumulated it as an attribute
    # (e.g. a GIFBuilder's `.frames`). Tolerate a mid-script error (the model may
    # save to a bad path AFTER building) — keep whatever was produced first.
    # Bundle modules resolve via cwd on sys.path.
    ns = {}
    code = _strip_fences(script)
    try:
        exec(code, ns)
    except Exception:
        pass  # partial ns retained: vars set before the failing line survive
    val = ns.get(varname)
    if not val and ("build_" + varname) in ns:
        try:
            val = ns["build_" + varname]()
        except Exception:
            val = None
    if not val:
        for v in ns.values():
            attr = getattr(v, varname, None)
            if isinstance(attr, list) and attr:
                val = attr
                break
    return val if val else []

def _exec_script_try(script, args=None):
    # Run the authored script but DON'T raise — return {ok, error, path} so the
    # node can hand a failure back to the model (repair_script) and retry, instead
    # of crashing the whole agent. The error text is the model's repair signal.
    try:
        p = _exec_script(script, args)
        return {"ok": True, "error": "", "path": p}
    except Exception as e:
        return {"ok": False, "error": str(e), "path": ""}

def _run_program(runner, target_file, content):
    # CHANGE 3 author-run atom: an authoring tool whose command is `<runner> <bare-file>`
    # (e.g. `node program.js`) receives the program CONTENT, not a path. Write the content
    # to the target, run it, non-raising. Returns {ok, error, out} so an author-loop can
    # re-author on failure (the discarded-output / node-content bugs from hand-tuning).
    try:
        with open(target_file, "w") as f:
            f.write(_strip_fences(str(content)))
    except Exception as e:
        return {"ok": False, "error": "write failed: " + str(e)[:200], "out": ""}
    try:
        r = _subprocess.run([str(runner), str(target_file)], capture_output=True, text=True, timeout=180)
        return {"ok": r.returncode == 0, "error": (r.stderr or "")[:800], "out": (r.stdout or "")}
    except Exception as e:
        return {"ok": False, "error": str(e)[:400], "out": ""}

def _run_cmd(prefix, args):
    # EMBODIMENT LADDER rung 2: the tool's IR carried a `command` (a bundle-script/CLI
    # invocation) but no inline body. Run it deterministically (owner=code) THROUGH THE
    # SHELL: authored commands carry `&&`/`|` and `{param}` placeholders, and argv
    # execution turned those into literal tokens (observed live: `/usr/bin/cd` ran with
    # `&&` as an argument and `{target_dir}` never substituted). args are (name, value)
    # pairs: a value substitutes shell-quoted into its `{name}` placeholder; values with
    # no placeholder append at the end; a missing value leaves ''. Bare (un-named) args
    # still append, so older call shapes keep working. Non-raising; returns combined
    # stdout+stderr so a downstream verify can read it.
    import shlex
    cmd = " ".join(str(x) for x in prefix) if isinstance(prefix, (list, tuple)) else str(prefix)
    extra = []
    for a in (args or []):
        k, v = (a if isinstance(a, (list, tuple)) and len(a) == 2 else (None, a))
        ph = ("{%s}" % k) if k else None
        if ph and ph in cmd:
            cmd = cmd.replace(ph, shlex.quote(str(v)) if v is not None else "''")
        elif v is not None:
            extra.append(shlex.quote(str(v)))
    if extra:
        cmd = cmd + " " + " ".join(extra)
    # Debian images ship python3 with no `python` alias, but a model drafts
    # "python x.py" by habit — the artifact then dies with
    # "[Errno 2] No such file or directory: 'python'", which reads as a task
    # failure. Rewrite only the COMMAND WORD, and only when python is absent.
    cmd = _py_fix(cmd)
    _tui_sub("$", _C, _fmt_val(cmd, 88))
    try:
        # `shell=True` runs /bin/sh — dash on Debian/Ubuntu — while a model drafts
        # BASH. dash then dies on the first "(" with exit 2 and the artifact reports
        # a failure that looks like a task error. Observed live on the SkillsBench
        # geospatial skill: three identical `/bin/sh: Syntax error: "(" unexpected`
        # before the agent gave up. Name the interpreter instead of inheriting it.
        r = _subprocess.run(cmd, shell=True, capture_output=True, text=True,
                            timeout=180, executable=_SHELL)
        out = ((r.stdout or "") + (r.stderr or "")).strip()
        _tui_sub("↳", _D, _fmt_val(out), "exit %s" % r.returncode)
        return out
    except Exception as e:
        _tui_sub("✗", "\033[31m", _fmt_val(e, 56))
        return "[cmd %s failed: %s]" % (cmd[:120], str(e)[:200])

_HAS_PY = __import__("shutil").which("python") is not None
_PY3 = __import__("shutil").which("python3") or "python3"

def _py_fix(cmd):
    if _HAS_PY:
        return cmd
    import re as _re
    # command position only: start of string, or after ; | && || ( newline
    return _re.sub(r"(?:(?<=^)|(?<=[;|&(\n]))(\s*)python(?=\s)",
                   lambda m: m.group(1) + _PY3, cmd)

_SHELL = __import__("shutil").which("bash") or "/bin/sh"

def _looks_python_file(p):
    import os
    try:
        q = str(p or "")
        if not (os.path.isfile(q) and not os.access(q, os.X_OK)):
            return False
        with open(q, "rb") as f:
            head = f.read(600)
        return (b"import " in head) or (b"def " in head) or head.startswith(b"\x23!") or (b"print(" in head)
    except Exception:
        return False

def _fix_argv(args):
    # Every subprocess a tool body spawns passes through here.
    #  * `python` on a Debian image with no alias -> the image's python3
    #    (court-form-filling: FileNotFoundError 'python' at the first node)
    #  * [sys.executable, "script.py"]: this interpreter is jac's EMBEDDED
    #    python, which has none of the task's libraries -> run task scripts on
    #    the image's python3 instead (exam-block-sequencing: pyscipopt missing
    #    although installed in the image). `-m pip` stays on the embedded one,
    #    that is the preflight installing modules for in-process import.
    #  * a generated, non-executable script used as the COMMAND WORD
    #    (energy-ac: `script_file.out: command not found`, exit 127) -> prefix
    #    the interpreter instead of failing.
    import os, sys as _s
    try:
        if isinstance(args, (list, tuple)) and args:
            a = [str(x) for x in args]
            if a[0] == "python" and not _HAS_PY:
                return [_PY3] + a[1:]
            if a[0] == _s.executable or a[0].endswith("/jac") or a[0] == "jac":
                if "-m" in a[:3] and "pip" in a[:4]:
                    return args
                if any(x.endswith(".py") or x.endswith(".out") or x == "-c" for x in a[1:3]):
                    return [_PY3] + a[1:]
                return args
            if _looks_python_file(a[0]):
                return [_PY3] + a
            return args
        if isinstance(args, str):
            t = args.strip()
            if not _HAS_PY:
                t = _py_fix(t)
            first = t.split()[0] if t.split() else ""
            if _looks_python_file(first):
                t = _PY3 + " " + t
            return t
    except Exception:
        pass
    return args

def _is_pip_install(args):
    # `python -m pip install ...`, `pip install ...`, `pip3 install ...` — list or string form
    try:
        s = [str(x) for x in args] if isinstance(args, (list, tuple)) else str(args).split()
        if not s:
            return False
        # pip must be the COMMAND (`pip install`, `/venv/bin/pip3 install`) or the
        # module an interpreter runs (`python3 -m pip install`) — never a word
        # inside some other command (`echo pip install`)
        if s[0].rsplit("/", 1)[-1] in ("pip", "pip3") and "install" in s[1:3]:
            return True
        if len(s) > 3 and s[1] == "-m" and s[2] in ("pip", "pip3") and "install" in s[3:5]:
            return True
    except Exception:
        pass
    return False

_PIP_LOG = {}
def _pip_log():
    # one append-mode log per program process; opened lazily, never raises
    if "fh" not in _PIP_LOG:
        import os as _o, tempfile as _t
        _PIP_LOG["path"] = _o.path.join(_t.gettempdir(), "sigil_pip_%d.log" % _o.getpid())
        _PIP_LOG["fh"] = open(_PIP_LOG["path"], "a")
    return _PIP_LOG["fh"]

_orig_popen_init = _subprocess.Popen.__init__
def _popen_init(self, args, *a, **kw):
    fixed = _fix_argv(args)
    # An installer the caller did not capture writes pip's own diagnostics — on a
    # failed build a ~20K-char internal traceback — straight into the terminal of
    # whoever called the program, and an agent calling it reads every byte
    # (SkillsBench travel: 20,124 of 27,206 chars per program call). Route that
    # output to a log instead; the caller still gets the return code, so its own
    # error handling and its one-line status are unchanged. Output the caller DID
    # capture (stdout/stderr passed, or run(capture_output=True)) is never touched.
    if (not a) and ("stdout" not in kw) and ("stderr" not in kw) and _is_pip_install(fixed):
        try:
            lg = _pip_log()
            kw["stdout"] = lg
            kw["stderr"] = lg
        except Exception:
            pass
    return _orig_popen_init(self, fixed, *a, **kw)
# a global patch: SIGIL_SUBPROCESS_FIX=0 leaves Popen exactly as python ships it
if _sigil_on("SUBPROCESS_FIX"):
    _subprocess.Popen.__init__ = _popen_init


_re_num = __import__("re").compile(r"-?\d+(?:\.\d+)?")

def _as_num(v, dflt, want_int=True):
    # Total coercion for an int/float carry — the same class as _as_dict. int()/float()
    # raise on None and on "" , which is why these carries were left at "carry keeps
    # default"; but dropping them silently breaks any counter or budget a route guards
    # on (a burn-down loop can never terminate, or never start). Parse what is
    # parseable, keep the default otherwise, never raise.
    try:
        if isinstance(v, bool):
            n = int(v)
        elif isinstance(v, (int, float)):
            n = v
        else:
            s = str(v).strip()
            m = _re_num.search(s)
            if not m:
                return dflt
            n = float(m.group(0))
        return int(n) if want_int else float(n)
    except Exception:
        return dflt

def _restructure(v, _depth=0):
    # byLLM cannot schematize a bare `dict`, so the lowering widens such a slot to
    # `dict[str, str]` — which forces EVERY value to a string. A spec whose values are
    # nested (`states: ["T1","T2"]`, `f: ["-a*T1 + u1", ...]`) then arrives as the REPR
    # of the list, and the consuming tool parses it as a scalar: sympify("['T1',...")
    # -> "could not parse '['", float("[") -> ValueError. Both runtimes hit it, so it is
    # structural, not a model fault (live: r2r-mpc-control both arms, exoplanet glm).
    # Restore the structure the model actually produced: a string that parses as a JSON
    # or Python list/dict becomes that list/dict. Scalars and prose are left alone —
    # only bracket/brace-delimited strings are even attempted.
    if _depth > 6:
        return v
    if isinstance(v, dict):
        return {k: _restructure(x, _depth + 1) for k, x in v.items()}
    if isinstance(v, (list, tuple)):
        return [_restructure(x, _depth + 1) for x in v]
    if isinstance(v, str):
        s = v.strip()
        if len(s) >= 2 and s[0] in "[{" and s[-1] in "]}":
            p = None
            try:
                p = _json.loads(s)
            except Exception:
                try:
                    import ast as _ast
                    p = _ast.literal_eval(s)
                except Exception:
                    p = None
            if isinstance(p, (list, dict)):
                return _restructure(p, _depth + 1)
    return v

def _as_dict(v, dflt):
    # Total coercion for a dict-typed carry. An Unknown py-block return must not clobber
    # the carry with garbage — but silently KEEPING the default severed the whole
    # downstream dataflow (observed live: synthesize_requirement ran every node on
    # inputs={} and hallucinated ungrounded prose). Accept a real mapping, parse a
    # JSON-object string, otherwise keep the default. Nested values that the
    # dict[str, str] widening flattened into reprs are restored (_restructure).
    if isinstance(v, dict):
        return _restructure(v)
    if isinstance(v, str):
        try:
            p = _json.loads(v)
        except Exception:
            p = None
        if isinstance(p, dict):
            return _restructure(p)
    return dflt

def _live_tool(name, args):
    # EMBODIMENT LADDER rung 0: a live-integration tool (reads the outside world) has no
    # scriptable body. First try a REGISTERED MCP server (Sigil wires SIGIL_MCP with the
    # user's servers); then a REAL web GET; otherwise an honest sentinel (never a hallucinated
    # script, so a grounded verify can tell it wasn't real).
    try:
        import sigil_mcp
        _m = sigil_mcp.dispatch(name, list(args) if args else [])
        if _m is not None:
            return _m["text"] if _m.get("ok") else "[mcp tool %s failed: %s]" % (name, _m.get("error", ""))
    except Exception:
        pass
    a0 = str(args[0]) if args else ""
    is_web = a0.startswith("http") or any(k in name for k in ("fetch", "_api", "_get", "web_", "search"))
    if is_web:
        url = a0 if a0.startswith("http") else ""
        if not url:
            return "[live tool %s: no URL argument provided]" % name
        try:
            import urllib.request
            with urllib.request.urlopen(url, timeout=30) as r:
                return r.read().decode("utf-8", "replace")[:100000]
        except Exception as e:
            return "[live fetch %s failed: %s]" % (name, str(e)[:200])
    return "[live source '%s' unavailable in the batch harness]" % name

# ---- sibling skills as tools: the library, reachable from inside a run ----
# Sigil hands a run the OTHER compiled skills it is allowed to use through
# SIGIL_SKILL_TOOLS (JSON; written by sigil_runtime.execute_module, already
# filtered by the operator's tool policy and with THIS skill removed so a
# procedure cannot call itself into a loop). Each becomes a byLLM tool on the
# autonomy slots that already carry tools, so a compiled procedure can hand a
# sub-job to a compiled procedure instead of re-deriving it.
#
# Sibling runs go out to their own `jac run` subprocess — the same isolation the
# parent got, for the same reason: one interpreter per run, no shared graph, no
# way for a callee's crash to take the caller down.
#
# No env (an ejected artifact, a bench workdir, any harness that is not Sigil) =
# an empty list, and the slot keeps exactly the tools it was compiled with.
_SKILL_TOOL_CACHE = None

def _skill_run(mod, task, timeout=900):
    import os, subprocess, sys, tempfile
    marker = "<<<SIGIL_REPORT>>>"
    body = (
        "import from crystallized.%s { run }\n"
        "import from jaclang.byllm.lib { Model }\n"
        "import os;\n"
        "with entry:__main__ {\n"
        "    m = Model(model_name=os.getenv(\"SIGIL_MODEL\", \"gpt-4o-mini\"));\n"
        "    out = run(os.getenv(\"SIGIL_TASK\", \"\"), m, \"\");\n"
        "    print(\"%s\");\n"
        "    print(out);\n"
        "}\n"
    ) % (mod, marker)
    fd, path = tempfile.mkstemp(suffix=".jac", prefix="_skilltool_", dir=os.getcwd())
    try:
        with os.fdopen(fd, "w") as f:
            f.write(body)
        env = dict(os.environ)
        env["SIGIL_TASK"] = str(task or "")
        # A nested run must not inherit the manifest, or skill A calling skill B
        # hands B the same shelf and the depth is bounded only by the timeout.
        env.pop("SIGIL_SKILL_TOOLS", None)
        env.pop("AGIR_PROGRESS", None)   # the parent's node trace is not the child's
        # --no-serve: jac >=0.37 serves a file near a web-app project by default
        p = subprocess.run([sys.executable, "run", "--no-serve", path], cwd=os.getcwd(), env=env,
                           capture_output=True, text=True, timeout=timeout)
    except subprocess.TimeoutExpired:
        return "[skill %s timed out after %ss]" % (mod, timeout)
    except Exception as e:
        return "[skill %s failed to start: %s]" % (mod, str(e)[:200])
    finally:
        try:
            os.unlink(path)
        except Exception:
            pass
    out = p.stdout or ""
    if marker in out:
        return out.split(marker, 1)[1].strip() or "[skill %s produced no report]" % mod
    return "[skill %s failed: %s]" % (mod, (p.stderr or out)[-400:])

def _skill_tools():
    global _SKILL_TOOL_CACHE
    if _SKILL_TOOL_CACHE is not None:
        return list(_SKILL_TOOL_CACHE)
    _SKILL_TOOL_CACHE = []
    raw = os.environ.get("SIGIL_SKILL_TOOLS", "")
    if not raw:
        return []
    try:
        from jaclang.byllm.lib import Tool
        rows = _json.loads(raw)
    except Exception:
        return []
    for row in rows if isinstance(rows, list) else []:
        try:
            name = str(row["name"])
            mod = str(row["module"])
            desc = str(row.get("description") or name)

            def _mk(_mod, _name, _desc):
                def _call(task=""):
                    return _tui_tool(_name, [("task", task)], lambda t=task: _skill_run(_mod, t))
                _call.__name__ = _name
                _call.__qualname__ = _name
                _call.__annotations__ = {"task": str, "return": str}
                _call.__doc__ = _desc
                _call._jac_semstr = _desc
                _call._jac_semstr_inner = {"task": "The sub-task to hand to this skill, in plain language."}
                return Tool(func=_call)

            _SKILL_TOOL_CACHE.append(_mk(mod, name, desc))
        except Exception:
            continue
    return list(_SKILL_TOOL_CACHE)

def _scoped_author_run(context, out_path=None, tries=3):
    # EMBODIMENT LADDER rung 3 (compiler redesign): the IR gave this tool no runnable
    # body, so instead of a silent `pass`, author a SCOPED script for exactly this tool's
    # job, run it, verify the artifact appeared, and re-author on failure. Non-raising.
    # The author is the module-level `_gen_scoped` by-llm slot (emitted whenever the
    # compiler falls back at least one tool); scoped = it sees only this job's context.
    import os
    gen = globals().get("_gen_scoped")
    if gen is None:
        return out_path or ""
    err = ""
    for _ in range(max(1, int(tries))):
        ctx = context if not err else (
            context + "\n\nThe previous script failed with:\n" + err +
            "\nReturn a corrected, COMPLETE, self-contained script.")
        try:
            script = _strip_fences(str(gen(ctx)))
        except Exception as e:
            err = "author error: " + str(e)[:300]
            continue
        res = _exec_script_try(script, None)
        ok = res.get("ok") and ((not out_path) or (os.path.exists(out_path) and os.path.getsize(out_path) > 0))
        if ok:
            return out_path or res.get("path") or "op.py"
        err = (res.get("error") or "the script produced no output file")[-500:]
    return out_path if (out_path and os.path.exists(out_path)) else ""

def _clean_path(p):
    # GEN-FILL slots (byLLM) occasionally leak JSON structure into a string field,
    # e.g. output_spec.path == 'out.txt},' or 'out.txt},"format":"txt"'. Trusting it
    # verbatim writes the artifact to a garbage filename and the deliverable is lost.
    # Strip trailing structural junk and cut at the first filename-illegal char after
    # a recognizable extension, so 'out.txt},...' -> 'out.txt'.
    import re
    s = str(p or "").strip().strip('`"\' ')
    s = re.sub(r'[}\]\)>,"\'`\s]+$', '', s)              # trailing }, ] ) , " ' ` ws
    m = re.match(r'^(.*?\.[A-Za-z0-9]{1,6})(?:[^A-Za-z0-9._/\\-].*)?$', s)
    if m:
        s = m.group(1)                                   # keep up to the extension
    return s

def _emit(data, args):
    spec = args.output_spec
    if getattr(spec, "destination", "") == "stdout":
        print(data)
        return "stdout"
    path = _clean_path(spec.path)
    ext = os.path.splitext(path)[1].lower().lstrip(".")
    fmt = (ext or getattr(spec, "format", "") or "txt").lower()
    if fmt == "json":
        with open(path, "w") as f:
            _json.dump(data, f, default=str)
    elif fmt == "xlsx":
        import pandas as pd
        frames = [pd.DataFrame(t[1:], columns=t[0]) for t in data if t]
        (pd.concat(frames, ignore_index=True) if frames else pd.DataFrame()).to_excel(path, index=False)
    elif fmt == "csv":
        import csv as _csv
        tables = data if isinstance(data, list) else [data]
        table = next((t for t in tables if t), [])
        with open(path, "w", newline="", encoding="utf-8") as f:
            w = _csv.writer(f)
            for row in table:
                w.writerow([(c if c is not None else "") for c in row])
    else:
        with open(path, "w") as f:
            f.write(data if isinstance(data, str) else str(data))
    return path

def _unwrap(ret, key, idx=0, single=True):
    # Resolve ONE write-carry from a tool's return. LIFT-authored bodies return
    # {"carry_name": value, ...}; legacy bodies return the bare value; a sig like
    # `-> (report, status)` returns a tuple bound positionally over the node's
    # writes. Total by design: a miss yields "" (never None, never a raise) so the
    # walker's typed coercions — str()/list()/bool() — stay safe on any shape.
    if isinstance(ret, dict):
        if key in ret:
            v = ret[key]
            return "" if v is None else v
        if single and len(ret) == 1:
            v = next(iter(ret.values()))
            return "" if v is None else v
        return ret if single else ""
    if isinstance(ret, (list, tuple)) and not single:
        return ret[idx] if idx < len(ret) else ""
    if single or idx == 0:
        return "" if ret is None else ret
    return ""

def _as_param(raw, default):
    """Bind a caller-supplied param to a typed walker carry.

    The MCP inputSchema declares every key as a string, so a list/dict/int input
    arrives as JSON text. Non-str `role: input` carries used to be dropped from
    the param surface entirely and kept their type defaults -- the walker then
    ran on [] / {} / 0 and, because empty-input checks pass vacuously, reported
    success while doing nothing.
    """
    if raw is None or raw == "":
        return default
    if not isinstance(raw, str):
        return raw
    import json as _pj
    try:
        v = _pj.loads(raw)
    except Exception:
        return default
    if isinstance(default, bool):
        return bool(v)
    if isinstance(default, (int, float)) and isinstance(v, (int, float)) and not isinstance(v, bool):
        return type(default)(v)
    if isinstance(default, (list, dict)):
        return v if isinstance(v, type(default)) else default
    return v


def _cap_seq(v, n=400, chars=60000):
    if not _sigil_on("INPUT_CAPS"):
        return v
    # A list/dict carry holding a whole DATA SET (exoplanet: the light curve as
    # {"time": [...5000 floats], "flux": [...]}) went into a decision slot verbatim:
    # 801,120 tokens on a 128k model, every downstream tool errored. Keep the
    # shape (a list stays a list, a dict a dict) but at most `n` elements per
    # sequence and ~`chars` of JSON overall; a decision needs a sample, not the set.
    import json as _j
    def _cap(x, depth=0):
        if isinstance(x, list):
            return [_cap(e, depth + 1) for e in x[:n]] + ([("... %d more" % (len(x) - n))] if len(x) > n else [])
        if isinstance(x, dict):
            out = {}
            for i, (k, val) in enumerate(x.items()):
                if i >= n:
                    out["..."] = "%d more keys" % (len(x) - n)
                    break
                out[k] = _cap(val, depth + 1)
            return out
        if isinstance(x, str) and len(x) > 4000:
            return x[:4000] + "... [truncated]"
        return x
    try:
        if not isinstance(v, (list, dict)):
            return v
        c = _cap(v)
        try:
            if len(_j.dumps(c, default=str)) <= chars:
                return c
        except Exception:
            return c
        # still too big: halve the per-sequence cap until it fits (floor 20)
        m = n
        while m > 20:
            m = m // 2
            n = m
            c = _cap(v)
            try:
                if len(_j.dumps(c, default=str)) <= chars:
                    return c
            except Exception:
                return c
        return c
    except Exception:
        return v

def _cap_in(v, n=60000):
    # A model slot fed a whole data file (exoplanet: the light-curve CSV went
    # into a GEN-ENUM decision) blows the context window; nothing an enum or a
    # plan needs lives past the first 60k chars. Code nodes never pass here.
    if not _sigil_on("INPUT_CAPS"):
        return v
    try:
        if isinstance(v, str) and len(v) > n:
            return v[:n] + "\n... [input truncated: %d more chars]" % (len(v) - n)
    except Exception:
        pass
    return v


# ---- runtime thinking-effort knob (SIGIL_THINK) -----------------------------
# glm-5.2 thinks before every typed answer; with the default effort a bool/int
# slot spent its whole budget thinking and the arm collected 17 typed-output
# failures. byLLM has no per-model setting for this and the call sites are
# compiled, so the knob is applied at the litellm boundary: SIGIL_THINK=low
# (or medium/high) -> reasoning_effort; SIGIL_THINK=false -> think=False.
# Only reasoning-model families are touched; litellm drops unknown params for
# the rest (drop_params is on in byLLM's config).
def _install_think_hook():
    import os
    mode = os.environ.get("SIGIL_THINK", "").strip().lower()
    if mode in ("", "off", "default"):
        return ""
    try:
        import litellm
    except Exception:
        return ""
    fam = ("glm", "deepseek", "qwen3", "qwq", "gpt-oss", "magistral")
    def _inject(kw):
        mode = os.environ.get("SIGIL_THINK", "").strip().lower()
        if not mode:
            return kw
        m = str(kw.get("model", "")).lower()
        if any(f in m for f in fam) and "think" not in kw and "reasoning_effort" not in kw:
            if mode in ("false", "no", "none"):
                kw["think"] = False
            else:
                kw["reasoning_effort"] = mode
        return kw
    try:
        _oc = litellm.completion
        def _c(*a, **kw):
            return _oc(*a, **_inject(kw))
        litellm.completion = _c
        _oa = litellm.acompletion
        async def _ac(*a, **kw):
            return await _oa(*a, **_inject(kw))
        litellm.acompletion = _ac
    except Exception:
        return ""
    return mode
_SIGIL_THINK = _install_think_hook()

_NOTHINK_STACK = []

def _nothink_push():
    # a str slot that came back EMPTY under a reasoning mode is retried once without
    # thinking: glm-5.2 at reasoning_effort=low returned "" for a whole solver script
    # (exam-block: an empty model file -> "no JSON payload"), think=False returned it
    _NOTHINK_STACK.append(os.environ.get("SIGIL_THINK"))
    os.environ["SIGIL_THINK"] = "false"

def _nothink_pop():
    prev = _NOTHINK_STACK.pop() if _NOTHINK_STACK else None
    if prev is None:
        os.environ.pop("SIGIL_THINK", None)
    else:
        os.environ["SIGIL_THINK"] = prev


_INPUT_PATHS = set()
_DEST_PATHS = set()   # paths the task names after a write/output verb
_WRITTEN_PATHS = set()   # files THIS run produced: never "inputs" even when re-read later

def _note_written_path(p):
    import os
    try:
        q = str(p or "").strip()
        if q:
            _WRITTEN_PATHS.add(os.path.abspath(q))
            _WRITTEN_PATHS.add(os.path.basename(q))
    except Exception:
        pass

def _note_input_path(p):
    # Files the caller handed to this run (or the task-named input). A write
    # target must never resolve onto one of them: _outname once scraped the
    # task's INPUT `/root/network.json` as the deliverable and the tool
    # overwrote the case file, killing every later tool in the cell (caught
    # live: energy-unit-commitment, both runtimes, 0/14).
    import os
    try:
        q = str(p or "").strip()
        # a file the run itself wrote and later re-reads (xlsx's repair_file.py in its
        # build/verify loop) is NOT an input: registering it made the next write to it
        # a "refused: INPUT" message, which the build step then executed as a command
        if q and (os.path.abspath(q) in _WRITTEN_PATHS or os.path.basename(q) in _WRITTEN_PATHS):
            return
        if q:
            _INPUT_PATHS.add(os.path.abspath(q))
            _INPUT_PATHS.add(os.path.basename(q))
    except Exception:
        pass

def _note_task_files(task, out_path=None):
    # Files the task text names that ALREADY EXIST are inputs by definition —
    # the run did not create them. _outname once scraped `/root/test.bib` (the
    # bibliography under review) as the write target for a derived .bib and the
    # validator overwrote the file it was checking (citation-check, glm, 9/9 -> 1/9).
    # The caller's own output path is exempt (in-place deliverables stay legal).
    import os, re
    try:
        outp = os.path.abspath(str(out_path)) if out_path else ""
        seen = set()
        for m in re.finditer(r'((?:/[A-Za-z0-9_.~-]+)+\.[A-Za-z0-9]{1,6}|\b[A-Za-z0-9_-]+(?:\.[A-Za-z0-9_-]+)*\.[A-Za-z0-9]{1,5})\b', str(task or "")):
            cand = m.group(1)
            if cand in seen:
                continue
            seen.add(cand)
            if not (os.path.isfile(cand) and os.path.abspath(cand) != outp):
                continue
            # A path the task names as the DESTINATION is not an input, even when a
            # file already sits there (a stub, or a previous run's output). Registering
            # it made the write guard refuse the deliverable itself: "Write your script
            # at: /root/workspace/solution.py" became an INPUT, the artifact wrote
            # solution.py.out instead, and the verifier found nothing
            # (crystallographic-wyckoff, both runtimes).
            head = str(task or "")[max(0, m.start() - 60):m.start()].lower()
            # the gap may carry "your script at:" or "the result to" — only a sentence
            # break (a period or newline) severs the verb from the path it governs
            if re.search(r"(write|save|output|emit|store|produce|create|put|place|export)\b[^.\n]{0,48}$", head):
                # Remember it as a DESTINATION. Declining to register it here is
                # not enough: the parameter binding registers the same file as an
                # input, so the write guard still refused the deliverable and the
                # refusal text was executed as a command downstream.
                try:
                    _DEST_PATHS.add(os.path.abspath(cand))
                    _DEST_PATHS.add(os.path.basename(cand))
                except Exception:
                    pass
                continue
            _note_input_path(cand)
    except Exception:
        pass

def _is_input_path(p):
    import os
    q = str(p or "").strip()
    if not q:
        return False
    # a path the task names as the destination wins over the input guard:
    # in-place deliverables ("Output answers in test-supply.xlsx") are legal
    if os.path.abspath(q) in _DEST_PATHS or os.path.basename(q) in _DEST_PATHS:
        return False
    return os.path.abspath(q) in _INPUT_PATHS or os.path.basename(q) in _INPUT_PATHS

def _outname(task, fallback="output.txt"):
    # GEN-RAW deliverables must land on the filename the TASK asks for (e.g. "save as
    # artifact.html", "Output to issues_out.json"), not a generic <carry>.out. Pull the
    # filename that follows a save/output/write verb; else the first bare filename; else fallback.
    # EXTENSION-AWARE: the fallback name encodes WHICH deliverable this write is
    # ("html_file.out", "philosophy_md_file.out") — when a task names several files
    # (philosophy.md AND sketch.html), extension-blind first-match sent EVERY export
    # to the same path (caught live: the HTML artifact wrote over philosophy.md).
    # Prefer candidates whose extension matches an ext token in the fallback.
    import re
    t = task or ""
    want = None
    fl = (fallback or "").lower()
    for ext in ("html", "pdf", "docx", "xlsx", "pptx", "csv", "json", "md", "gif", "png", "svg", "py", "js", "txt"):
        if re.search(r'(?:^|[_.-])' + ext + r'(?:$|[_.-])', fl):
            want = ext
            break
    cands = []
    # No leading \b on the capture: a word boundary can never precede the '/' of
    # an absolute path, so the match started one char late and "/app/out/x.json"
    # came back as "app/out/x.json" -- then written, relative to cwd, to
    # /app/app/out/x.json. _inname below already carries this stem.
    # The verb must START a word that is not a path component: "Save result to
    # `/output/scenario_3.json`" matched `output` inside the path and captured
    # "/scenario_3.json" — a root-level file the run could not create (civ6:
    # "wrote (not written) instead"). Any task with an /output/ directory hit it.
    m = re.search(r'(?<![/\w])(?:save|output|write|export|emit|produce)\b[^.\n]{0,60}?([A-Za-z0-9_.~/-]*[A-Za-z0-9_]\.[A-Za-z0-9]{1,5})\b', t, re.I)
    if m:
        cands.append(m.group(1))
    cands += re.findall(r'([A-Za-z0-9_.~/-]*[A-Za-z0-9_]\.(?:html?|pdf|docx|xlsx|pptx|csv|json|txt|md|gif|png|jpe?g|svg|tsx?|jsx?|py|js|npy|npz|nc|parquet|h5))\b', t, re.I)
    cands = [c for c in cands if not _is_input_path(c)]
    if want:
        for c in cands:
            ce = c.rsplit(".", 1)[-1].lower()
            if ce == want or (want == "html" and ce == "htm"):
                return c
        # a typed write must never land on a foreign-extension path
        return re.sub(r'[^A-Za-z0-9_.-]', '_', (fl.rsplit('.', 1)[0] or 'output')) + "." + want
    if cands:
        return cands[0]
    return fallback

def _inname(task, fallback=""):
    # The SENSE mirror of _outname — NOT a reuse: _outname's first preference is
    # the save-verb filename, which on the read side is exactly the wrong file
    # ("clean data.csv and save as out.csv" must read data.csv, never out.csv).
    # Prefer the filename following a read-side verb; else the first bare
    # filename NOT claimed by a save-side verb; else fallback ("" keeps the
    # tool's own guard error honest when the task truly names no input).
    # No leading \b on the captures: a word boundary can never precede the '/'
    # of an absolute path (or the '~' of a home path), so \b silently strips
    # the leading slash — "read /a/b.json" must return "/a/b.json", never
    # "a/b.json". The stems accept full paths for the same reason: the bare
    # fallback matching only basenames plucked "explainer.json" out of the
    # MIDDLE of a pathname.
    import re
    t = task or ""
    m = re.search(r'(?:read|open|load|parse|clean|process|convert|analy[sz]e|summari[sz]e|merge|from)\b[^.\n]{0,60}?([A-Za-z0-9_.~/-]*[A-Za-z0-9_]\.[A-Za-z0-9]{1,5})\b', t, re.I)
    if m:
        return m.group(1)
    saved = set()
    for sm in re.finditer(r'(?:save|output|write|export|emit|produce)\b[^.\n]{0,60}?([A-Za-z0-9_.~/-]*[A-Za-z0-9_]\.[A-Za-z0-9]{1,5})\b', t, re.I):
        saved.add(sm.group(1))
    for bm in re.finditer(r'([A-Za-z0-9_.~/-]*[A-Za-z0-9_]\.(?:html?|pdf|docx|xlsx|pptx|csv|json|txt|md|gif|png|jpe?g|svg|tsx?|jsx?|py|js))\b', t, re.I):
        if bm.group(1) not in saved:
            return bm.group(1)
    return fallback

def _extract_json(text):
    # A str slot asked for a JSON deliverable often wraps the document in a code
    # fence or a sentence ("Here is the report:\n{...}"). Return the canonical
    # JSON text when ONE whole document can be recovered, else None. Never invents.
    import json as _j, re as _re
    t = str(text or "").strip()
    if not t:
        return None
    for cand in (t, _strip_fences(t)):
        try:
            return _j.dumps(_j.loads(cand), indent=2, ensure_ascii=False)
        except Exception:
            pass
    m = _re.search(r"```(?:json)?\s*([\[{].*?[\]}])\s*```", t, _re.S)
    if m:
        try:
            return _j.dumps(_j.loads(m.group(1)), indent=2, ensure_ascii=False)
        except Exception:
            pass
    # outermost document first: whichever bracket opens earlier in the text
    pairs = sorted((("{", "}"), ("[", "]")), key=lambda pc: (t.find(pc[0]) < 0, t.find(pc[0])))
    for o, c in pairs:
        i = t.find(o)
        k = t.rfind(c)
        if i >= 0 and k > i:
            try:
                return _j.dumps(_j.loads(t[i:k + 1]), indent=2, ensure_ascii=False)
            except Exception:
                pass
    return None

def _json_guard(path, text):
    # -> (text_to_write, recovered?). A .json path names a FORMAT, not just a
    # location: recover a fenced/prose-wrapped document, else None so the caller
    # parks the text in a .md sibling and says so (an absent deliverable is
    # recoverable by the agent; a corrupt one is not).
    import os as _o, json as _j
    t = text if isinstance(text, str) else str(text)
    if _o.path.splitext(str(path or ""))[1].lower() != ".json" or not t.strip():
        return (t, False)
    try:
        _j.loads(t)
        return (t, False)
    except Exception:
        pass
    ex = _extract_json(t)
    if ex is not None:
        return (ex, True)
    return (None, False)

def _refuse_json_write(path, text):
    import os as _o, sys as _s
    _alt = _o.path.splitext(str(path))[0] + ".md"
    try:
        _write_artifact_raw(text, _alt)
    except Exception:
        _alt = ""
    if _alt:
        return ("[write_file refused: %s must contain JSON but this content is not JSON, so it was saved "
                "as %s instead. If the task needs JSON at %s, produce it yourself from that file.]" % (path, _alt, path))
    return ("[write_file refused: %s must contain JSON but this content is not JSON, and the fallback file "
            "could not be written either (check the directory). Produce the JSON at %s yourself.]" % (path, path))

_TASK_TEXT = {"t": ""}

def _note_task_text(task):
    # The task text is the only place a required OUTPUT SCHEMA is written down, and a
    # skill's tools are generic by design, so nothing downstream can check the shape it
    # produced against the shape the task asked for.
    _TASK_TEXT["t"] = str(task or "")
    return task

def _schema_gap(path, content):
    # When the task shows a JSON example for the file being written, name the top-level
    # keys it declares that the content does not carry. energy-market-pricing asks for
    # {"base_case": ..., "counterfactual": ...}; the LMP skill wrote its own generic
    # report and reported success, and the verifier failed on a missing base_case.
    import json as _j, re as _re
    try:
        if not str(path or "").lower().endswith(".json"):
            return ""
        task = _TASK_TEXT.get("t") or ""
        if not task:
            return ""
        want = []
        for blk in _re.findall(r"```(?:json)?\s*(\{[\s\S]*?\})\s*```", task):
            probe = _re.sub(r"\.\.\.", "", blk)
            try:
                obj = _j.loads(probe)
            except Exception:
                obj = None
                for k in _re.findall(r'^\s{2}"([A-Za-z0-9_]+)"\s*:', blk, _re.M):
                    if k not in want:
                        want.append(k)
            if isinstance(obj, dict):
                for k in obj:
                    if k not in want:
                        want.append(k)
        if not want:
            return ""
        try:
            got = _j.loads(str(content or ""))
        except Exception:
            return ""
        if not isinstance(got, dict):
            return ""
        missing = [k for k in want if k not in got]
        if not missing:
            return ""
        return (" [schema gap: the task specifies %s at the top level of this file; wrote %s. "
                "Produce the missing key(s) yourself before delivering.]"
                % (", ".join(want), ", ".join(sorted(got)) or "an empty object"))
    except Exception:
        return ""

def _format_contract(task):
    # The deliverable FORMAT is part of the contract every slot must honour: a
    # str slot that "authors the report" wrote Markdown into the report.json the
    # task asked for (3d-scan, azure-bgp, fix-visual-stability, civ6 — both
    # runtimes) and the write was refused. Say it once, in the task text that
    # every slot reads. Names no file, so the _outname/_inname scrapes are unmoved.
    import re as _re
    t = str(task or "")
    if (not _sigil_on("WRITE_GUARDS")) or "[DELIVERABLE FORMAT]" in t:
        return t
    names = [n for n in _re.findall(r'([A-Za-z0-9_.~/-]*[A-Za-z0-9_]\.json)\b', t) if not _is_input_path(n)]
    if not names:
        return t
    return (t + "\n\n[DELIVERABLE FORMAT] Every deliverable FILE whose filename ends in .json must contain "
            "exactly one valid JSON document and nothing else: no Markdown, no prose, no code fences, "
            "no comments. Use exactly the keys and structure the task specifies. This concerns the file's "
            "contents only: intermediate steps (drafts, entries, notes, analyses) keep their own natural format.")


def _csv_guard(path, text):
    # A .csv deliverable written with a Markdown report (gravitational-wave: the
    # matched-filter REPORT landed in detection_results.csv, "Missing required
    # columns. Found: ['# Matched Filter Report']") is the CSV twin of the JSON
    # refusal. -> (text_to_write, recovered?) or (None, False) to refuse. A
    # fenced ```csv block inside prose is recovered; never invents rows.
    import os as _o, re as _re
    t = text if isinstance(text, str) else ("" if text is None else str(text))
    if _o.path.splitext(str(path or ""))[1].lower() != ".csv" or not t.strip():
        return (t, False)
    def _looks_csv(u):
        lines = [l for l in u.strip().splitlines() if l.strip()]
        if not lines:
            return False
        head = lines[0].lstrip()
        if head.startswith("#") or head.startswith("```"):
            return False
        return any(d in head for d in (",", "\t", ";", "|"))
    if _looks_csv(t):
        return (t, False)
    m = _re.search(r"```(?:csv)?\s*\n(.*?)```", t, _re.S)
    if m and _looks_csv(m.group(1)):
        return (m.group(1).strip() + "\n", True)
    return (None, False)

def _refuse_csv_write(path, text):
    import os as _o
    _alt = _o.path.splitext(str(path))[0] + ".md"
    try:
        _write_artifact_raw(text, _alt)
    except Exception:
        _alt = ""
    if _alt:
        return ("[write_file refused: %s must contain CSV rows but this content is a report/prose, so it was saved "
                "as %s instead. If the task needs CSV at %s, produce the rows yourself from that file.]" % (path, _alt, path))
    return "[write_file refused: %s must contain CSV rows but this content is a report/prose; produce the CSV at %s yourself.]" % (path, path)

def _write_artifact(content, path):
    if not _sigil_on("WRITE_GUARDS"):
        return _write_artifact_raw(content, path)
    # EMPTY content never creates or overwrites a deliverable: an upstream carry that
    # came out "" (citation-management: format_bibtex found no entries) was written
    # over the task's answer.json as a 0-byte file — worse than no file, since the
    # caller then sees the deliverable "present". Say so instead.
    _c0 = content if isinstance(content, str) else ("" if content is None else str(content))
    if not _c0.strip():
        return "[write skipped: nothing to write for " + str(path) + " (empty content); the file was not created or changed]"
    # never clobber a file the caller handed in as INPUT (see _note_input_path)
    if _is_input_path(path):
        import os as _o
        _alt = str(path) + ".out"
        _write_artifact.__dict__["_redirected"] = _alt
        return "[write refused: " + str(path) + " is an INPUT of this run; wrote " + _alt + " instead] " + str(_write_artifact_raw(content, _alt))
    _bin_ext = (".xlsx", ".xlsm", ".xls", ".docx", ".pptx", ".pdf", ".png", ".jpg", ".jpeg", ".gif", ".zip", ".parquet", ".h5", ".npz", ".sqlite", ".db", ".civ6map")
    if os.path.splitext(str(path))[1].lower() in _bin_ext:
        # text can never be a valid binary document: park it beside the path instead
        _alt1 = str(path) + ".txt"
        try:
            _write_artifact_raw(_c0, _alt1)
        except Exception:
            _alt1 = "(not written)"
        return ("[write refused: %s is a binary document format and this content is text; saved it as %s instead. "
                "Produce the real file with the proper library.]" % (path, _alt1))
    _txt, _rec = _json_guard(path, content)
    if _txt is None:
        return _refuse_json_write(path, content if isinstance(content, str) else str(content))
    _txt, _rec2 = _csv_guard(path, _txt)
    if _txt is None:
        return _refuse_csv_write(path, content if isinstance(content, str) else str(content))
    _rec = _rec or _rec2
    if _rec:
        import sys as _s
        print("[write: recovered the JSON document from surrounding text for " + str(path) + "]", file=_s.stderr)
    _out = _write_artifact_raw(_txt, path)
    # a generic skill cannot know a task's required report shape; say so rather than
    # reporting a bare path the caller will trust
    return str(_out) + _schema_gap(path, _txt)

def _write_artifact_raw(content, path):
    os.makedirs(os.path.dirname(path) or ".", exist_ok=True)
    with open(path, "w") as f:
        f.write(content if isinstance(content, str) else str(content))
    _note_written_path(path)
    return path

def _artifact_text(v, limit=8000):
    # Surface the DELIVERED CONTENT, not just a path. Delivery ops return the path
    # of the file the authored script wrote/printed; the caller wants the ANSWER, not
    # the filename. If v names a TEXT file, return its (stripped, truncated) content;
    # a BINARY file (pdf/img/xlsx/…) is surfaced as "path (N bytes)" not raw bytes;
    # anything that isn't a file is returned as-is. Never raises.
    try:
        s = v if isinstance(v, str) else str(v)
        if s and os.path.isfile(s):
            _bin = (".pdf", ".png", ".jpg", ".jpeg", ".gif", ".webp", ".xlsx", ".docx",
                    ".pptx", ".zip", ".mp4", ".mp3", ".bin", ".parquet")
            if s.lower().endswith(_bin):
                return "%s (%d bytes)" % (s, os.path.getsize(s))
            with open(s, "r", errors="replace") as f:
                data = f.read().strip()
            return data[:limit] + ("…[truncated]" if len(data) > limit else "")
        return s
    except Exception:
        return str(v)

def _emit_surface(content, spec):
    # An ACT·artifact surface (sig only, no IR body) must still WRITE the deliverable
    # to an app-controlled path — a bodyless surface lowered to `pass` silently drops
    # the output (the internal-comms/etc. 0% bug). `spec` may be an ArtifactSpec (has
    # .path/.destination/.format), a carrier that holds one (OpArgs.output_spec), or a dict.
    if spec is not None and hasattr(spec, "output_spec"):
        spec = spec.output_spec
    def _g(o, k, d=None):
        if o is None: return d
        if isinstance(o, dict): return o.get(k, d)
        return getattr(o, k, d)
    text = content if isinstance(content, str) else ("" if content is None else str(content))
    if (_g(spec, "destination", "") or "") == "stdout":
        print(text); return "stdout"
    path = _clean_path(_g(spec, "path", "") or "")
    if not path:
        path = "output." + str(_g(spec, "format", "") or "txt").lstrip(".")
    return _write_artifact(text, path)

def _read_module(relpath):
    here = os.path.dirname(os.path.abspath(__file__))
    for cand in (relpath, os.path.join(here, relpath)):
        if os.path.exists(cand):
            with open(cand) as f:
                return f.read()
    return ""
